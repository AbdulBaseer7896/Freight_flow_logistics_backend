============================================================
FREIGHT FLOW BACKEND - SERVER SETUP GUIDE
============================================================

1. INSTALL POSTGRESQL & SYSTEM DEPENDENCIES
------------------------------------------------------------
sudo apt update
sudo apt install -y postgresql postgresql-contrib libpq-dev python3-pip python3-venv

sudo systemctl start postgresql
sudo systemctl enable postgresql


2. CREATE DATABASE & USER
------------------------------------------------------------
sudo -u postgres psql

-- Inside psql:
CREATE DATABASE freight_db;
CREATE USER freight_user WITH PASSWORD 'YourStrongPassword123!';
ALTER ROLE freight_user SET client_encoding TO 'utf8';
ALTER ROLE freight_user SET default_transaction_isolation TO 'read committed';
ALTER ROLE freight_user SET timezone TO 'UTC';
GRANT ALL PRIVILEGES ON DATABASE freight_db TO freight_user;
ALTER DATABASE freight_db OWNER TO freight_user;
\c freight_db
GRANT ALL ON SCHEMA public TO freight_user;
\q


3. PYTHON VIRTUAL ENVIRONMENT & PACKAGES
------------------------------------------------------------
cd /opt/Freight_flow_logistics_backend
python3 -m venv venv
source venv/bin/activate
pip install --upgrade pip
pip install -r requirements.txt


4. CONFIGURE .ENV
------------------------------------------------------------
nano .env

# Put the following inside .env:
SECRET_KEY=your-secret-key-here
ENVIRONMENT=production
DEBUG=False
ALLOWED_HOSTS=*
DATABASE_URL=postgresql://freight_user:YourStrongPassword123!@127.0.0.1:5432/freight_db
CORS_ALLOWED_ORIGINS=https://your-cpanel-frontend.com,https://www.your-cpanel-frontend.com


5. RUN MIGRATIONS & CREATE ADMIN
------------------------------------------------------------
python manage.py migrate
python manage.py createsuperuser


6. CREATE SYSTEMD SERVICE (FOR BACKGROUND AUTO-START)
------------------------------------------------------------
sudo nano /etc/systemd/system/freightflow.service

# Paste the following into the service file:
[Unit]
Description=Freight Flow Backend Gunicorn Daemon
After=network.target postgresql.service

[Service]
User=root
Group=root
WorkingDirectory=/opt/Freight_flow_logistics_backend
ExecStart=/opt/Freight_flow_logistics_backend/venv/bin/gunicorn --workers 3 --bind 127.0.0.1:8000 backend.wsgi:application
Restart=always
RestartSec=3

[Install]
WantedBy=multi-user.target


7. START & ENABLE THE BACKEND SERVICE
------------------------------------------------------------
sudo systemctl daemon-reload
sudo systemctl start freightflow
sudo systemctl enable freightflow

# Check service status:
sudo systemctl status freightflow

# View live logs:
journalctl -u freightflow -f


8. CONFIGURE NGINX (REVERSE PROXY)
------------------------------------------------------------
sudo apt install -y nginx certbot python3-certbot-nginx

sudo nano /etc/nginx/sites-available/freightflow

# Paste the following into the file:
# (Replace 178-128-148-182.sslip.io with your actual domain or IP hostname)
server {
    server_name 178-128-148-182.sslip.io;
    client_max_body_size 50M;

    location /uploads/ {
        alias /opt/Freight_flow_logistics_backend/uploads/;
        expires 30d;
        add_header Cache-Control "public, no-transform";
        add_header Access-Control-Allow-Origin *;
    }

    location / {
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}

# Enable the Nginx site configuration:
sudo ln -s /etc/nginx/sites-available/freightflow /etc/nginx/sites-enabled/
sudo rm -f /etc/nginx/sites-enabled/default
sudo nginx -t
sudo systemctl restart nginx


9. INSTALL FREE HTTPS SSL (CERTBOT)
------------------------------------------------------------
sudo certbot --nginx -d 178-128-148-182.sslip.io

# Auto-renewal check:
sudo certbot renew --dry-run
