cat > /etc/nginx/sites-available/freight <<'EOF'
server {
    listen 80;
    server_name 143.110.250.205;
    client_max_body_size 50M;
    location / {
        proxy_pass http://127.0.0.1:1000;
        proxy_http_version 1.1;
